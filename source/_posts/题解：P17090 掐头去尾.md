# 题解：P17090 掐头去尾

## 思路

我们注意到，双端队列在任意时刻总有：队首总是当前前缀最大值，队尾总是当前最小值。因此长度增加的唯一情形是新元素同时不超过队首和队尾，也就是它成为新的全局最小值。反过来，如果新元素成为新的前缀最大值，它只会替换队首，长度不变。

对原序列每个位置 $i$，记 $L[i]$ 为左边第一个不小于 $a_i$ 的位置（不存在则为 $0$），用单调递减栈可以预处理出来。固定区间 $[l,r]$，当扫描到 $i\in(l,r]$ 时，若 $L[i]<l$，说明从 $l$ 到 $i-1$ 中没有任何元素大于等于 $a_i$，也就是 $a_i$ 严格大于之前所有元素，它是新的前缀最大值，不会让队列变长。反之，若 $L[i]\ge l$，则它在左侧有一个“靠山”，才有可能成为新的最小值而追加到队尾。我们把这类满足 $L[i]\ge l$ 的位置称为有效位置。

将所有有效位置按原序连成一条链。第一个有效位置出现时，队列长度从 $1$ 变成 $2$。之后相邻两个有效位置 $p$ 和 $q$（$p<q$），如果 $a_q\le a_p$，说明 $a_q$ 比当前队尾还小，从队尾压入，长度加 $1$；否则它只会替换队尾，长度不变。所以区间答案就是：若链为空则为 $1$，否则为 $2$ 加上链上所有满足“后一个元素不大于前一个元素”的相邻对数。

离线扫描左端点 $l$。当左端点从 $l$ 变成 $l+1$ 时，原本有效的条件从 $L[i]\ge l$ 变成 $L[i]\ge l+1$，因此恰好所有满足 $L[i]=l$ 的位置会从有效链中删除。预处理时按 $L[i]$ 把儿子挂成邻接表，扫描过程中每向右移动一步就把这些点删掉。

删除时用双向链表维护有效链的前后关系，用树状数组维护每个有效点与其前驱之间的贡献标记：若当前点比前驱小则标记为 $1$，否则为 $0$。删除一个点后，只需要清空它的贡献，并重新计算它的后继与新前驱之间的贡献关系。同时用并查集压缩路径，跳过已被删除的位置，方便快速找到当前左端点之后第一个仍然有效的点。

回答询问时，先通过并查集找到 $l+1$ 之后第一个有效点 $p_1$。如果 $p_1>r$，说明区间内没有有效点，答案为 $1$。否则直接查树状数组在 $(p_1+1,r)$ 内的贡献和，再加上固定的 $2$ 就是最终队列长度。总复杂度 $O((n+q)\log n)$。

## 代码

给一个场上码风极丑的代码。

```cpp
#include <bits/stdc++.h>
#define endl '\n'
using namespace std;

const int MAXN = 500005;
int n, q;
int a[MAXN];
int L[MAXN];
int stk[MAXN], stktop;
int toth[MAXN], totn[MAXN];
int qh[MAXN], qn[MAXN];
int ql[MAXN], qr[MAXN];
int ans[MAXN];
int prep[MAXN + 2], nxtp[MAXN + 2];
int bv[MAXN];
int fen[MAXN];
int par[MAXN + 2];

static char inbuf[1 << 20];
int inlen = 0, inpos = 0;
inline int gc()
{
    if (inpos == inlen)
    {
        inlen = fread(inbuf, 1, 1 << 20, stdin);
        inpos = 0;
        if (!inlen)
            return -1;
    }
    return inbuf[inpos++];
}
inline int readInt()
{
    int c = gc();
    while (c != '-' && (c < '0' || c > '9'))
        c = gc();
    int sgn = 1;
    if (c == '-')
    {
        sgn = -1;
        c = gc();
    }
    int x = 0;
    while (c >= '0' && c <= '9')
    {
        x = x * 10 + c - '0';
        c = gc();
    }
    return x * sgn;
}

static char outbuf[1 << 23];
int outpos = 0;
inline void writeInt(int x)
{
    if (x == 0)
    {
        outbuf[outpos++] = '0';
    }
    else
    {
        char tmp[12];
        int t = 0;
        while (x)
        {
            tmp[t++] = '0' + x % 10;
            x /= 10;
        }
        while (t--)
            outbuf[outpos++] = tmp[t];
    }
    outbuf[outpos++] = '\n';
}

inline void upd(int i, int delta)
{
    for (; i <= n; i += i & (-i))
        fen[i] += delta;
}
inline int qry(int i)
{
    int s = 0;
    for (; i > 0; i -= i & (-i))
        s += fen[i];
    return s;
}
inline int range(int l, int r)
{
    if (l > r)
        return 0;
    if (l < 1)
        l = 1;
    return qry(r) - qry(l - 1);
}

int find(int x)
{
    int root = x;
    while (par[root] != root)
        root = par[root];
    while (par[x] != root)
    {
        int nx = par[x];
        par[x] = root;
        x = nx;
    }
    return root;
}

void rep(int i)
{
    int p = prep[i], qq = nxtp[i];
    upd(i, -bv[i]);
    bv[i] = 0;
    nxtp[p] = qq;
    prep[qq] = p;
    if (qq <= n)
    {
        int newbit = 0;
        if (p >= 1 && p <= n)
            newbit = (a[qq] <= a[p]) ? 1 : 0;
        upd(qq, newbit - bv[qq]);
        bv[qq] = newbit;
    }
    par[i] = find(i + 1);
}

main()
{
    n = readInt();
    q = readInt();
    for (int i = 1; i <= n; i++)
        a[i] = readInt();

    stktop = 0;
    for (int i = 1; i <= n; i++)
    {
        while (stktop > 0 && a[stk[stktop]] < a[i])
            stktop--;
        L[i] = stktop > 0 ? stk[stktop] : 0;
        stk[++stktop] = i;
    }

    for (int i = 0; i <= n; i++)
        toth[i] = 0;
    for (int i = 1; i <= n; i++)
    {
        totn[i] = toth[L[i]];
        toth[L[i]] = i;
    }

    int lst = 0;
    for (int i = 1; i <= n; i++)
    {
        if (L[i] > 0)
        {
            prep[i] = lst;
            nxtp[lst] = i;
            lst = i;
            if (prep[i] != 0)
                bv[i] = (a[i] <= a[prep[i]]) ? 1 : 0;
            else
                bv[i] = 0;
            par[i] = i;
        }
        else
        {
            par[i] = i + 1;
            bv[i] = 0;
        }
    }
    nxtp[lst] = n + 1;
    prep[n + 1] = lst;
    par[n + 1] = n + 1;

    for (int i = 1; i <= n; i++)
    {
        if (bv[i])
            upd(i, bv[i]);
    }

    for (int i = 1; i <= n; i++)
        qh[i] = 0;
    for (int j = 1; j <= q; j++)
    {
        int l = readInt(), r = readInt();
        ql[j] = l;
        qr[j] = r;
        qn[j] = qh[l];
        qh[l] = j;
    }

    for (int l = 1; l <= n; l++)
    {
        if (l >= 2)
        {
            int v = l - 1;
            for (int i = toth[v]; i != 0; i = totn[i])
                rep(i);
        }
        int p1 = find(l + 1);
        for (int j = qh[l]; j != 0; j = qn[j])
        {
            int r = qr[j];
            if (p1 > r)
                ans[j] = 1;
            else
            {
                int base = 2;
                int s = range(p1 + 1, r);
                ans[j] = base + s;
            }
        }
    }

    for (int j = 1; j <= q; j++)
        writeInt(ans[j]);
    fwrite(outbuf, 1, outpos, stdout);
    return 0;
}
```
